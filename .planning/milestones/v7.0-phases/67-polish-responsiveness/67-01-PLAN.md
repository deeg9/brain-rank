Create `web/src/components/landing/ScrollReveal.tsx` with this exact implementation (per DES-03, 21st.dev FadeInOnScroll pattern using Framer Motion useInView):

```typescript
'use client';
import { motion, useInView } from 'framer-motion';
import { useRef } from 'react';

interface ScrollRevealProps {
  children: React.ReactNode;
  className?: string;
  delay?: number;
  direction?: 'up' | 'down';
}

export function ScrollReveal({ children, className = '', delay = 0, direction = 'up' }: ScrollRevealProps) {
  const ref = useRef(null);
  const isInView = useInView(ref, { once: true, margin: '-100px' });
  
  return (
    <motion.div
      ref={ref}
      initial={{ opacity: 0, y: direction === 'up' ? 40 : -40 }}
      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: direction === 'up' ? 40 : -40 }}
      transition={{ duration: 0.7, delay, ease: [0.25, 0.25, 0.25, 0.75] }}
      className={className}
    >
      {children}
    </motion.div>
  );
}
```
