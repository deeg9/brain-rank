   ```typescript
   })
   ```

3. Create `web/src/lib/supabase/admin.ts` with:
   ```typescript
   import { createClient } from '@supabase/supabase-js'
   import type { Database } from '@/lib/database.types'

   // Service role client -- bypasses RLS. Use ONLY in server-side code (webhook handlers, admin operations).
   // NEVER import this in client components or expose SUPABASE_SERVICE_ROLE_KEY to the browser.
   export const supabaseAdmin = createClient<Database>(
     process.env.NEXT_PUBLIC_SUPABASE_URL!,
     process.env.SUPABASE_SERVICE_ROLE_KEY!,
     {
       auth: {
         autoRefreshToken: false,
         persistSession: false,
       },
     }
   )
   ```

   Import `createClient` from `@supabase/supabase-js` (NOT `@supabase/ssr`). The admin client does not use cookies/sessions -- it authenticates purely via the service role key.

   Update the apiVersion string in stripe.ts to match whatever `Stripe.LatestApiVersion` resolves to, so TypeScript does not error on version mismatch (per PROD-01: no hardcoded IDs, and the SDK version pin must be consistent).
