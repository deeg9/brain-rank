"use client";

import { getInitials, getInitialsColor } from "@/lib/avatar";

interface PublicAvatarDisplayProps {

export function PublicAvatarDisplay({ username }: PublicAvatarDisplayProps) {
  return (
    <div
      className={`w-full h-full rounded-full flex items-center justify-center ${getInitialsColor(username, username)}`}
    >
      <span className="text-white text-2xl font-bold">
        {getInitials(username, username)}
      </span>
    </div>
  );
}

Create `web/src/app/u/[username]/page.tsx` as an async server component. This route is OUTSIDE the `(app)` layout group so it has no auth requirement. Per PROF-02 and context decisions.

**Data fetching (use Supabase server client -- works without auth due to RLS allowing public read):**

1. Extract `username` from `params.username`
2. Query profiles table: `.from("profiles").select("id, username, avatar_url, bio, created_at").eq("username", username).single()`
3. If no profile found, call `notFound()` from `next/navigation` (triggers Next.js 404)
4. Cast result as `{ id: string; username: string; avatar_url: string | null; bio: string | null; created_at: string }`
5. Query public jobs: `.from("jobs").select("*").eq("user_id", profile.id).eq("is_public", true).eq("status", "completed").order("created_at", { ascending: false })`
6. Cast result as `Job[]` from `@/lib/database.types`

**Stats computation (server-side, same pattern as private profile):**
- `totalPublicAnalyses` = publicJobs.length
- Extract `overall_score` from each job's `results` object (same cast pattern as private profile: `(j.results as Record<string, unknown> | null)?.overall_score`)
- `averageScore` = mean of scores, formatted to 1 decimal, or "--" if no scores
- `bestScore` = max of scores, formatted to 1 decimal, or "--" if no scores

**Member since:**
- Derive from `profile.created_at`: `new Date(profile.created_at).toLocaleDateString("en-US", { month: "long", year: "numeric" })`

**Layout (mirror private profile page structure with these differences):**
- No "Edit Profile" button (this is a public view)
- No email displayed (privacy)
- Stats show "Public Analyses" instead of "Total Analyses"

**Hero section:**
- Avatar: 96x96 circle with neural-gradient border. If `profile.avatar_url` exists, render `<img>`. Otherwise render `<PublicAvatarDisplay username={profile.username} />`
- Username: `<h2>` with `@{profile.username}`
- Bio: paragraph if exists
- Member since line

**Stats bento grid (3 columns on md, 1 on mobile):**
- "Public Analyses" count
- "Average Score" in text-secondary
- "Best Score" in text-primary-fixed

**Public analyses list:**
- Header: "Public Analyses"

- Each job card: movie icon, title (from `job.title ?? job.video_url.split("/").pop() ?? "Untitled"`), date, neural score badge showing `overall_score` formatted to 1 decimal (or "N/A")
- Each card is NOT a link yet (public analysis view is Phase 48) -- just a display card
- Use same styling as private profile's recent activity cards: `bg-surface-container-low rounded-lg` with hover transition
