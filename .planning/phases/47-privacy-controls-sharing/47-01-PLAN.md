The dashboard is a server component, so the visibility toggle on cards needs a client component wrapper. Create an inline client component within the dashboard or extract to a small file.

1. Create a new client component file `web/src/app/(app)/dashboard/VisibilityToggle.tsx`:
   ```typescript
   "use client";
   import { useState } from "react";

   export function VisibilityToggle({ jobId, initialPublic }: { jobId: string; initialPublic: boolean }) {
     const [isPublic, setIsPublic] = useState(initialPublic);
     const [updating, setUpdating] = useState(false);

     async function handleToggle() {
       setUpdating(true);
       const newValue = !isPublic;
       setIsPublic(newValue); // optimistic
       try {
         const res = await fetch(`/api/jobs/${jobId}`, {
           method: "PATCH",
           headers: { "Content-Type": "application/json" },
           body: JSON.stringify({ is_public: newValue }),
         });
         if (!res.ok) setIsPublic(!newValue); // revert
       } catch {
         setIsPublic(!newValue); // revert
       }
       setUpdating(false);
     }

     return (
       <button
         onClick={(e) => { e.preventDefault(); e.stopPropagation(); handleToggle(); }}
         disabled={updating}
         className="flex items-center gap-1 text-xs text-on-surface-variant hover:text-on-surface transition-colors"
         title={isPublic ? "Public - click to make private" : "Private - click to make public"}
       >
         <span className="material-symbols-outlined text-sm">
           {isPublic ? "public" : "lock"}
         </span>
         {isPublic && (
           <span className="text-[10px] font-headline uppercase tracking-widest font-bold text-primary">Public</span>
         )}
       </button>
     );
   }
   ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-47-01 | Tampering | PATCH /api/jobs/[id] | mitigate | Validate is_public is boolean type before accepting; enforce ownership via .eq('user_id', user.id) on update query |
| T-47-02 | Information Disclosure | GET /api/jobs/[id] | mitigate | Private jobs return 404 (not 403) to non-owners to avoid leaking existence; public jobs omit RunPod polling (no internal infra details) |
| T-47-03 | Elevation of Privilege | PATCH /api/jobs/[id] | mitigate | Auth required for all PATCH requests; RLS + .eq('user_id') ensures users can only modify own jobs |
| T-47-04 | Information Disclosure | GET /api/jobs/[id] public path | accept | Public jobs intentionally expose results data (scores, segments) -- this is the feature. No PII beyond what user chose to share. |
