```typescript
export async function uploadText(text: string): Promise<string> {
  const supabase = createClient()
  const fileName = `${crypto.randomUUID()}.txt`
  const filePath = `uploads/${fileName}`
  const blob = new Blob([text], { type: 'text/plain' })

  const { error } = await supabase.storage
    .from('videos')
    .upload(filePath, blob, {
      cacheControl: '3600',
      upsert: false,
      contentType: 'text/plain',
    })

  if (error) throw new Error(`Upload failed: ${error.message}`)

  const { data: { publicUrl } } = supabase.storage
    .from('videos')
    .getPublicUrl(filePath)

  return publicUrl
}
```

Key details:
- Takes a `string` (the text content), NOT a File
- Creates a Blob with text/plain content type
- Uses the same 'videos' bucket and 'uploads/' prefix as image/video
- Returns the public URL of the uploaded .txt file
