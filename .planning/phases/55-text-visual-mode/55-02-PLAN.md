**3. Add handleAnalyzeTextVisual function** (near handleAnalyzeText, around line 131):
```typescript
async function handleAnalyzeTextVisual() {
  if (!textContent.trim()) return;
  setUploading(true);
  setError(null);

  try {
    const textUrl = await uploadText(textContent);
    const jobId = await createJob(textUrl, 'text_visual', textContent, textStyle);
    router.push(`/processing?jobId=${jobId}`);
  } catch (err) {
    setError(err instanceof Error ? err.message : 'Upload failed.');
    setUploading(false);
  }
}
```
