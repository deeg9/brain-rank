**Create `web/src/components/layout/AppShell.tsx`:**

Wraps authenticated pages with Sidebar + TopBar. Children render in the main content area.

```tsx
import { Sidebar } from "./Sidebar";
import { TopBar } from "./TopBar";

interface AppShellProps {
  children: React.ReactNode;
  topBarProps?: {
    title?: string;
    showSearch?: boolean;
    showBackArrow?: boolean;
    backHref?: string;
    rightActions?: React.ReactNode;
  };
}

export function AppShell({ children, topBarProps }: AppShellProps) {
  return (
    <>
      <Sidebar activeRoute="" />
      <TopBar {...topBarProps} />
      <main className="ml-64 pt-16 min-h-screen bg-surface p-8">
        <div className="max-w-7xl mx-auto">
          {children}
        </div>
      </main>
    </>
  );
}
```

The key layout values matching Stitch: `ml-64` (matches sidebar w-64), `pt-16` (matches topbar h-16), `bg-surface`, `p-8`, content constrained to `max-w-7xl`.
