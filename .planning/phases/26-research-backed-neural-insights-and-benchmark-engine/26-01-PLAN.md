```typescript
export type TierName = 'Elite' | 'Strong' | 'Average' | 'Weak' | 'Critical';
export type TierColor = 'tertiary' | 'primary' | 'error';

export interface Tier { tier: TierName; label: string; color: TierColor; }

export type HookType = 'strong_viral_hook' | 'engaging_unfocused' | 'attention_no_payoff' | 'weak_hook' | 'no_hook_data';
export interface HookAnalysis {
  type: HookType;
  headline: string;        // e.g. "Strong viral hook"
  description: string;     // e.g. "Your opening activates reward centers at 87% — that's stronger than 85% of viral content"
  reward: number;          // avg reward in 0-5s
  attention: number;       // avg attention in 0-5s
  tier: Tier;
}

export interface DropoffPoint {
  segmentIndex: number;
  timestamp: number;       // start time of the dropping segment
  drop: number;            // magnitude of drop (0-1)
  recommendation: string;  // e.g. "Add a pattern interrupt at 1:23"
}

export interface ShareMoment {
  segmentIndex: number;
  timestamp: number;
  score: number;           // social_cognition value
}

export interface FlowState {
  startTime: number;
  endTime: number;
  segmentCount: number;
  avgAttention: number;
  avgReward: number;
}

export interface ViralityScore {
  score: number;           // 0-1
  tier: Tier;
  components: {
    hookStrength: number;
    retention: number;
    emotionalArc: number;
    shareability: number;
  };
}

export interface Recommendation {
  icon: string;            // material-symbols-outlined name
  title: string;
  description: string;
  priority: 'high' | 'medium' | 'low';
}

export interface BenchmarkResult {
  virality: ViralityScore;
  hook: HookAnalysis;
  dropoffs: DropoffPoint[];
  shareMoments: ShareMoment[];
  flowStates: FlowState[];
  recommendations: Recommendation[];
}
```

**Functions per D-02 through D-07:**

`getTier(score: number): Tier` — Per D-02 thresholds exactly: >0.85 Elite/Top 10%/tertiary, 0.70-0.85 Strong/Top 25%/tertiary, 0.45-0.70 Average/Average/primary, 0.25-0.45 Weak/Below Average/error, <0.25 Critical/Bottom 10%/error.

`analyzeHook(segments: SegmentData[]): HookAnalysis` — Per D-03: filter segments where start < 5. Average their reward and attention. Apply the 4-quadrant classification using 0.70 threshold for both. If no segments in window, return type "no_hook_data" with a message about the video being too short. Set headline/description to feel like a neuroscience advisor (smart, specific). Include the actual scores in the description. Set tier from the average of reward + attention.

`detectDropoffs(segments: SegmentData[]): DropoffPoint[]` — Per D-04: iterate segments[1..n], compare each segment's emotional_processing to the previous. If drop > 30% (i.e., prev * 0.7 > current), record it. Include a concrete recommendation using formatTimestamp for the timestamp (import from @/lib/results), suggesting "pattern interrupt (cut, zoom, or music change)".

`detectShareMoments(segments: SegmentData[]): ShareMoment[]` — Per D-05: find local maxima of social_cognition where score > 0.70. A local maximum means the value is >= both its neighbors (handle first/last segment as having only one neighbor).

`detectFlowStates(segments: SegmentData[]): FlowState[]` — Per D-06: find contiguous runs where attention > 0.60 AND reward > 0.60. Merge adjacent qualifying segments into FlowState ranges with start/end times and averages.

`computeViralityScore(segments: SegmentData[], summary: SummaryData): ViralityScore` — Per D-07 exactly:
- hookStrength = average of (reward + attention) / 2 for segments where start < 5. If none, use summary.hook_score.
- retention = average attention across all segments. If none, use summary.retention_score.
- emotionalArc = variance of emotional_processing across segments (use population variance: sum of (x - mean)^2 / n). Normalize to 0-1 by capping at 0.1 variance = 1.0 (since variance of scores in [0,1] rarely exceeds 0.1).
- shareability = max social_cognition across all segments. If none, use 0.
- Weighted: score = hookStrength * 0.30 + retention * 0.25 + emotionalArc * 0.20 + shareability * 0.25.
- Map to tier via getTier.

`generateRecommendations(hook: HookAnalysis, dropoffs: DropoffPoint[], shareMoments: ShareMoment[], flowStates: FlowState[], virality: ViralityScore): Recommendation[]` — Produce 2-3 recommendations. Logic: if hook is weak/critical, recommend improving opener. If dropoffs exist, recommend pattern interrupts at specific timestamps. If no share moments, recommend adding relatable/social content. If no flow states, recommend tightening pacing. If virality is Average or below, pick the weakest component. Cap at 3 recommendations, prioritize by impact. Use "edit_note", "trending_up", "share", "psychology" as icons.

`analyzeBenchmark(segments: SegmentData[], summary: SummaryData): BenchmarkResult` — Top-level: calls all the above, returns unified result. Handle edge cases: empty segments array returns sensible defaults (all zeros, no moments, generic recommendation).
